# ======================================================================================
#  Корневой CMakeLists.txt репозитория
# ======================================================================================
#
#  В репозитории лежат два независимых примера, каждый в своей папке со своим
#  корневым CMakeLists.txt:
#
#      math-example/      — основы CMake: статическая библиотека и линковка
#      strategy-example/  — сторонние зависимости и паттерн Strategy
#
#  Каждый пример можно собрать отдельно, прямо как самостоятельный проект:
#
#      cmake -S math-example     -B build/math-example
#      cmake -S strategy-example -B build/strategy-example
#
#  Этот файл нужен для того, чтобы можно было собрать всё сразу из корня:
#
#      cmake -S . -B build && cmake --build build
#
#  Он не содержит ни одной собственной цели — только подключает подпапки.
# ======================================================================================

cmake_minimum_required(VERSION 3.16)

# Имя проекта-контейнера. Внутри подпапок вызывается свой project() —
# вложенные проекты в CMake разрешены: каждый из них задаёт собственные
# PROJECT_SOURCE_DIR и PROJECT_BINARY_DIR, но граф целей у них общий,
# поэтому имена целей во всём дереве должны быть уникальными.
project(HseProjectTemplate CXX)

# add_subdirectory читает CMakeLists.txt из указанной папки и выполняет его
# в дочерней области видимости: переменные, заданные внутри, наружу не протекают.
# Поэтому настройки одного примера не влияют на другой.
add_subdirectory(math-example)
add_subdirectory(strategy-example)
