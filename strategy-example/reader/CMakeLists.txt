# ======================================================================================
#  Библиотека reader: стратегии чтения файлов
# ======================================================================================

# Статическая библиотека из четырёх .cpp файлов (см. семинар про CMake и libmath.a).
add_library(reader STATIC
    src/json_strategy.cpp
    src/xlsx_strategy.cpp
    src/strategy_factory.cpp
    src/file_content.cpp
)

# Наши собственные заголовки: PUBLIC, потому что их включает main.cpp.
target_include_directories(reader SYSTEM PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# --------------------------------------------------------------------------------------
#  PUBLIC / PRIVATE / INTERFACE в target_link_libraries
# --------------------------------------------------------------------------------------
#  Ключевое слово отвечает на два вопроса: нужна ли зависимость при сборке самой цели
#  и нужна ли она тем, кто линкуется с целью (транзитивно).
#
#      PRIVATE    — нужна только при сборке reader. Дальше не передаётся.
#      PUBLIC     — нужна и при сборке reader, и всем, кто линкуется с reader.
#      INTERFACE  — самой reader не нужна, но нужна тем, кто с ней линкуется
#                   (так подключают header-only библиотеки, у которых нет .cpp).
#
#  Правило простое: смотрим на ПУБЛИЧНЫЕ ЗАГОЛОВКИ цели.
#  Если тип из сторонней библиотеки встречается в нашем .hpp (поле класса,
#  параметр, тип возврата) — зависимость PUBLIC: без её заголовков наш .hpp
#  не скомпилируется у пользователя. Если библиотека упоминается только в .cpp —
#  PRIVATE.
#
#  У нас заголовки reader/include/*.hpp не упоминают ни nlohmann::json,
#  ни OpenXLSX::XLDocument: интерфейс IReadStrategy описан через std::string.
#  Поэтому обе зависимости PRIVATE, и это даёт два практических эффекта:
#
#    1. main.cpp компилируется, не видя заголовков nlohmann/json и OpenXLSX;
#    2. замена OpenXLSX на другую xlsx-библиотеку не затрагивает ни одного файла
#       вне reader/src/xlsx_strategy.cpp.
#
#  Важно: PRIVATE прячет только ЗАГОЛОВКИ. Объектный код статической библиотеки
#  никуда не исчезает — линковщику всё равно нужны символы OpenXLSX при сборке
#  исполняемого файла, и CMake сам передаст libOpenXLSX.a в команду линковки
#  (это и есть транзитивность на уровне линковки, $<LINK_ONLY:...> в экспорте).
# --------------------------------------------------------------------------------------
target_link_libraries(reader
    PRIVATE
        nlohmann_json::nlohmann_json   # header-only: INTERFACE-цель, компилировать нечего
        OpenXLSX::OpenXLSX             # компилируемая библиотека: libOpenXLSX.a + miniz + pugixml
        project_warnings               # наша INTERFACE-цель с флагами предупреждений
)
